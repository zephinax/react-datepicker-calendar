      Example.tsx
      import DatePicker, { Calendar, utils } from '@zephinax/react-datepicker-calendar';

export function Example() {
  const [value, setValue] = useState(utils('en').getToday());
  return (
    <DatePicker
      value={value}
      onChange={setValue}
      inputPlaceholder="Choose a date"
      shouldHighlightWeekends
    />
  );
}
